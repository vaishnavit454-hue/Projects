Real Browser Bookmarklet

To clip highlights directly from any website into Margin while reading, save this bookmarklet to your browser's bookmarks bar:

javascript:(function(){const s=window.getSelection().toString().trim();if(!s){alert('Please highlight text first!');return;}const url=window.location.href;const clips=JSON.parse(localStorage.getItem('collider_clips')||'[]');clips.unshift({id:'clip_'+Date.now(),url:url,highlighted_text:s,date_saved:new Date().toISOString()});localStorage.setItem('collider_clips',JSON.stringify(clips));alert('Saved to Margin!');})();
